void LinBusLogic::handleMessage(cMessage *msg) {
    // We cast to cPacket because all our LIN types (Break, Header, Response)
    // inherit from cPacket (via the .msg definition)
    cPacket *frame = check_and_cast<cPacket *>(msg);

    EV_INFO << "Bus forwarding signal: " << frame->getName()
            << " (" << frame->getBitLength() << " bits)" << endl;

    // Broadcast to all connected nodes
    // Note: In a true electrical bus, signals go to everyone including sender.
    // Logic here forwards to all ports.
    int gateCount = gateSize("gate$o");
    for (int i = 0; i < gateCount; i++) {
        // We dup() for everyone except the last one to avoid memory leaks
        // (Standard OMNeT++ idiom)
        send(i == gateCount - 1 ? frame : frame->dup(), "gate$o", i);
    }
}

#ifndef LINBUSLOGIC_H_
#define LINBUSLOGIC_H_

#include <omnetpp.h>
using namespace omnetpp;

class LinBusLogic : public cSimpleModule {
    struct LinSlot {
        int frameId;
        std::string publisher;
        double period;
    };

    std::vector<LinSlot> scheduleTable;

    // === Step 1: Timing Parameters ===
    double baudRate;        // bits per second (e.g., 19200)
    double propDelay;       // propagation delay (seconds)
    double interByteTime;   // delay between bytes (seconds)

    // Utility to compute transmission time of given bit length
    double computeTxTime(int bits) {
        return (double)bits / baudRate;
    }

protected:
    virtual void initialize() override;
    virtual void handleMessage(cMessage *msg) override;
};

#endif

// 1. The Break field is a physical signal (dominant state), not a data byte.
// It signals the start of a frame.
packet LinBreakSignal {
    double duration; // Duration of the dominant state (min 13 bit times)
}

// 2. The Header remainder (Sync + PID) sent by the Master
packet LinHeader {
    int frameId;      // Useful for simulation logic (metadata)
    int pid;          // Protected Identifier (6-bit ID + 2 parity bits)
    char syncByte = 0x55; 
}

// 3. The Response part (Data + Checksum) sent by Publisher (Master or Slave)
packet LinResponse {
    int frameId;      // Metadata to help receivers match response to header
    int dataLength = 8;
    char data[8];
    char checksum;    // Classic or Enhanced checksum
}

#include "LinNode.h"
#include "LinUtil.h"

Define_Module(LinNode);

void LinNode::initialize() {
    isMaster = par("isMaster");
    baudRate = par("baudRate");
    propDelay = par("propDelay");
    interByteTime = par("interByteTime");

    state = IDLE;

    scheduleTimer = new cMessage("scheduleTimer");
    headerTimer = new cMessage("headerTimer");

    if (isMaster) {
        EV_INFO << "Master node initialized (State Machine Active)\n";

        // Define LIN schedule
        scheduleTable.push_back({0x10, "master", 0.02}); // Master publishes
        scheduleTable.push_back({0x12, "slave1", 0.02}); // Slave 1 publishes
        scheduleTable.push_back({0x14, "slave2", 0.02}); // Slave 2 publishes

        scheduleAt(simTime() + scheduleTable[0].period, scheduleTimer);
    } else {
        EV_INFO << "Slave node initialized (Listening).\n";
    }
}

void LinNode::handleMessage(cMessage *msg) {
    // -----------------------------------------------------
    // SELF MESSAGES (Internal Timers)
    // -----------------------------------------------------

    // 1. MASTER: Start of a new Schedule Slot
    if (msg == scheduleTimer) {
        LinSlot &slot = scheduleTable[currentSlotIndex];
        EV_INFO << ">>> Starting Slot for Frame ID 0x" << std::hex << slot.frameId << std::dec << endl;

        // Step 1: Send BREAK Signal
        sendBreakSignal();

        // Step 2: Schedule Header (Sync+PID) after Break duration
        // 13 bits for break + delimiter space
        double breakDuration = 13.0 / baudRate;
        scheduleAt(simTime() + breakDuration + interByteTime, headerTimer);

        // Schedule next slot
        scheduleAt(simTime() + slot.period, scheduleTimer);
        currentSlotIndex = (currentSlotIndex + 1) % scheduleTable.size();
        return;
    }

    // 2. MASTER: Send Sync + PID (Header Remainder)
    if (msg == headerTimer) {
        // Retrieve the ID for the slot we just started
        int idx = (currentSlotIndex == 0) ? scheduleTable.size() - 1 : currentSlotIndex - 1;
        int frameId = scheduleTable[idx].frameId;
        std::string publisher = scheduleTable[idx].publisher;

        sendHeaderRemainder(frameId, publisher);
        return;
    }

    // -----------------------------------------------------
    // NETWORK MESSAGES (Received from Bus)
    // -----------------------------------------------------

    // Case 1: BREAK SIGNAL Received
    if (dynamic_cast<LinBreakSignal*>(msg)) {
        EV_DETAIL << "Received BREAK signal. Resetting state to WAIT_FOR_SYNC.\n";
        state = WAIT_FOR_SYNC;
        delete msg;
        return;
    }

    // Case 2: HEADER Received
    if (dynamic_cast<LinHeader*>(msg)) {
        LinHeader *header = (LinHeader*)msg;

        // Validation
        if (state != WAIT_FOR_SYNC && state != IDLE) {
             EV_WARN << "Header received while not waiting for sync! State=" << state << endl;
        }

        processReceivedHeader(header);
        delete msg;
        return;
    }

    // Case 3: RESPONSE Received
    if (dynamic_cast<LinResponse*>(msg)) {
        LinResponse *rsp = (LinResponse*)msg;
        EV_INFO << "Received RESPONSE for Frame ID 0x" << std::hex << rsp->getFrameId()
                << ". Data: " << (char)rsp->getData(0) << "...\n";
        state = IDLE;
        delete msg;
        return;
    }

    delete msg; // Safely delete unexpected messages
}

void LinNode::sendBreakSignal() {
    LinBreakSignal *brk = new LinBreakSignal("LIN_Break");
    brk->setDuration(13.0 / baudRate);
    // Break is immediate
    send(brk, "gate$o");
}

void LinNode::sendHeaderRemainder(int frameId, std::string publisher) {
    LinHeader *header = new LinHeader("LIN_Header");
    header->setFrameId(frameId);
    header->setPid(computePID(frameId));
    header->setSyncByte(0x55);

    // Calculate Header Transmission Time (approx 20 bits: 10 for Sync, 10 for PID)
    double headerTxTime = 20.0 / baudRate;

    // 1. Send Header (Event occurs at receiver when header is FULLY received)
    sendDelayed(header, headerTxTime + propDelay, "gate$o");

    // 2. If Master is the publisher, it must send the Response too.
    // CRITICAL FIX: We must wait for the Header to finish (headerTxTime)
    // PLUS the Response Space (interByteTime) before starting the Response.
    if (publisher == "master") {
        double delayBeforeResponseStarts = headerTxTime + interByteTime;
        sendResponse(frameId, "MASTER_DATA", delayBeforeResponseStarts);
    }
}

void LinNode::processReceivedHeader(LinHeader *header) {
    state = WAIT_FOR_PID;
    int id = header->getFrameId();
    const char* myName = getName();

    // Check if I am the publisher
    bool iAmPublisher = false;
    if (strcmp(myName, "slave1") == 0 && id == 0x12) iAmPublisher = true;
    else if (strcmp(myName, "slave2") == 0 && id == 0x14) iAmPublisher = true;

    if (iAmPublisher) {
        EV_INFO << "I am the publisher for ID 0x" << std::hex << id << ". Sending response.\n";
        // Slave receives header at 'now'. So delay is just the Response Space.
        double delay = interByteTime;
        sendResponse(id, (strcmp(myName, "slave1") == 0) ? "SLAVE1_D" : "SLAVE2_D", delay);
    } else {
        state = RECEIVING;
    }
}

void LinNode::sendResponse(int frameId, const char* dataPattern, double initialDelay) {
    LinResponse *rsp = new LinResponse("LIN_Response");
    rsp->setFrameId(frameId);
    rsp->setDataLength(8);

    // Fill data
    for (int i=0; i<8; i++) {
        rsp->setData(i, (i < strlen(dataPattern)) ? dataPattern[i] : 0);
    }

    // Compute Checksum
    int pid = computePID(frameId);
    unsigned char sum = computeChecksum(rsp, pid);
    rsp->setChecksum(sum);

    // Timing:
    // initialDelay: Time to wait before STARTING transmission
    // (Master: headerTx + gap. Slave: gap)
    // responseTxTime: Time the bus is busy with the response (90 bits)
    double responseTxTime = 90.0 / baudRate;

    // The message arrives at the destination after (StartDelay + TxDuration + PropDelay)
    sendDelayed(rsp, initialDelay + responseTxTime + propDelay, "gate$o");

    state = TRANSMITTING;
}

void LinNode::finish() {
    cancelAndDelete(scheduleTimer);
    cancelAndDelete(headerTimer);
}

#ifndef LINNODE_H_

#include <omnetpp.h>
#include <vector>
#include <string>

using namespace omnetpp;

// ISO 17987 Node States
enum LinNodeState {
    IDLE,               // Bus is free
    WAIT_FOR_SYNC,      // Detected Break, waiting for 0x55
    WAIT_FOR_PID,       // Detected Sync, waiting for PID
    TRANSMITTING,       // Sending response (Publisher)
    RECEIVING           // Receiving response (Subscriber)
};

struct LinSlot {
    int frameId;
    std::string publisher;
    double period;
};

class LinNode : public cSimpleModule
{
private:
    bool isMaster;
    LinNodeState state;

    // Timers
    cMessage *scheduleTimer;   // Master only: triggers next slot
    cMessage *headerTimer;     // Master only: triggers Sync+PID after Break

    // Timing parameters
    double baudRate;
    double propDelay;
    double interByteTime;

    // Master schedule table
    std::vector<LinSlot> scheduleTable;
    int currentSlotIndex = 0;

    // Helpers
    void sendBreakSignal();
    // Update: added publisher arg
    void sendHeaderRemainder(int frameId, std::string publisher);
    // Update: added initialDelay arg
    void sendResponse(int frameId, const char* dataPattern, double initialDelay);
    void processReceivedHeader(LinHeader *header);

protected:
    virtual void initialize() override;
    virtual void handleMessage(cMessage *msg) override;
    virtual void finish() override;
};

#endif

--- FILE: src/LinUtil.cc ---
#include "LinUtil.h"

unsigned char computePID(int frameId)
{
    unsigned char id = frameId & 0x3F;

    // LIN 2.0+ Parity calculation
    // P0 = ID0 ^ ID1 ^ ID2 ^ ID4
    unsigned char p0 = ((id >> 0) ^ (id >> 1) ^ (id >> 2) ^ (id >> 4)) & 1;
    // P1 = ~(ID1 ^ ID3 ^ ID4 ^ ID5)
    unsigned char p1 = (~((id >> 1) ^ (id >> 3) ^ (id >> 4) ^ (id >> 5))) & 1;

    return id | (p0 << 6) | (p1 << 7);
}

unsigned char computeChecksum(LinResponse *frame, int pid)
{
    int len = frame->getDataLength();
    unsigned int sum = 0;

    // Check for Classic Checksum (used for diagnostics 0x3C, 0x3D)
    // or if the node is configured for LIN 1.x (not modeled here, assuming LIN 2.x)
    bool classic = (pid == 0x3C || pid == 0x3D);

    // Enhanced Checksum includes the PID
    if (!classic) {
        sum += pid;
    }

    // Add Data Bytes
    for (int i = 0; i < len; ++i) {
        sum += (unsigned char)frame->getData(i);
        // Handle carry immediately or after loop - LIN standard uses inverted modulo 256 with carry
        // Simple addition loop here, strict carry handling below
    }

    // Fold carries (add high byte to low byte)
    while (sum > 0xFF) {
        sum = (sum & 0xFF) + (sum >> 8);
    }

    // Invert result
    return (~sum) & 0xFF;
}
